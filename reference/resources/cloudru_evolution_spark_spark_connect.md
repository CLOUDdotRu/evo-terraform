# cloudru_evolution_spark_spark_connect (Resource)



## Example Usage

```terraform
resource "cloudru_evolution_spark_spark_connect" "resource_spark_connect" {
  # Поле spark_id является неизменяемым. При изменении значения ресурс будет пересоздан.
  spark_id = "0000-0000-0000-0000-0000"
  # Поле base_image является неизменяемым. При изменении значения ресурс будет пересоздан.
  # Нужно заполнить одно из значений - base_image, custom_image
  base_image = {
    # Поле id является неизменяемым. При изменении значения ресурс будет пересоздан.
    id = "0000-0000-0000-0000-0000"
  }
  # Поле custom_image является неизменяемым. При изменении значения ресурс будет пересоздан.
  custom_image = {
    custom_image_uri = "cr.cloud.ru/my-project/my-spark-image:latest"
  }
  configuration = {
    # Поле config_spark является неизменяемым. При изменении значения ресурс будет пересоздан.
    config_spark = {
      "spark.driver.extraJavaOptions" = "-Dlog4j.configuration=file:log4j.properties"
    }
  }
  # Поле server является неизменяемым. При изменении значения ресурс будет пересоздан.
  server = {
    # Поле cores является неизменяемым. При изменении значения ресурс будет пересоздан.
    cores = 1
    # Поле memory является неизменяемым. При изменении значения ресурс будет пересоздан.
    memory = "1g"
  }
  # Поле executor является неизменяемым. При изменении значения ресурс будет пересоздан.
  executor = {
    # Поле count является неизменяемым. При изменении значения ресурс будет пересоздан.
    count = 1
    # Поле cores является неизменяемым. При изменении значения ресурс будет пересоздан.
    cores = 1
    # Поле memory является неизменяемым. При изменении значения ресурс будет пересоздан.
    memory = "1g"
  }
  # Позволяет переопределить дефолтный таймаут провайдера для определенного метода. Если нужно указать бесконечный таймаут, то нужно указать например 0s, тогда таймаута не будет.
  timeouts {
    create = "60m"
    update = "30m"
    delete = "20m"
  }
  # Игнорировать изменения таймаутов: это предотвращает  лишние обновления ресурса при смене значений таймаутов в конфигурации
  # Но следует учитывать, что метод delete в ресурсе читает значение таймаута из стейта и, если его нужно изменить, то этот блок стоит закомментировать
  lifecycle {
    ignore_changes = [timeouts]
  }
}
```
