# cloudru_evolution_redis_cluster (Resource)



## Example Usage

```terraform
resource "cloudru_evolution_redis_cluster" "resource_cluster" {
  # Поле name является неизменяемым. При изменении значения ресурс будет пересоздан.
  name = "name-of-cluster_1"
  # Поле version_id является неизменяемым. При изменении значения ресурс будет пересоздан.
  version_id       = "00000000-0000-0000-0000-000000000000"
  specification_id = "00000000-0000-0000-0000-000000000000"
  # Поле type является неизменяемым. При изменении значения ресурс будет пересоздан.
  # Варианты значений параметра type:
  # STANDALONE, CLUSTER
  type = "STANDALONE"
  # Поле shards является неизменяемым. При изменении значения ресурс будет пересоздан.
  shards = 3
  # Поле replicas является неизменяемым. При изменении значения ресурс будет пересоздан.
  replicas    = 2
  storage_gib = 15
  # Поле logging является неизменяемым. При изменении значения ресурс будет пересоздан.
  logging = {
    enabled      = true
    log_group_id = "00000000-0000-0000-0000-000000000000"
  }
  # Поле subnet_ids является неизменяемым. При изменении значения ресурс будет пересоздан.
  subnet_ids = ["00000000-0000-0000-0000-000000000000", "00000000-0000-0000-0000-000000000000"]
  parameters = {
    "appendonly"                                   = "false",
    "client-output-buffer-limit-normal|hard-limit" = "1048576"
  }
  user_description = "Some short text."
  project_id       = "00000000-0000-0000-0000-000000000000"
  # Позволяет переопределить дефолтный таймаут провайдера для определенного метода. Если нужно указать бесконечный таймаут, то нужно указать например 0s, тогда таймаута не будет.
  timeouts {
    create = "60m"
    update = "30m"
    delete = "20m"
  }
  # Игнорировать изменения таймаутов: это предотвращает  лишние обновления ресурса при смене значений таймаутов в конфигурации
  # Но следует учитывать, что метод delete в ресурсе читает значение таймаута из стейта и, если его нужно изменить, то этот блок стоит закомментировать
  lifecycle {
    ignore_changes = [timeouts]
  }
}
```
